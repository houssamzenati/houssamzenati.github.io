% Match policy_learning_frames.tex: blue fitted objects, a red true target,
% pale green for estimation, and neutral text and axes.
\colorlet{ink}{black}
\colorlet{muted}{black!60}

% One local schematic: nearby nuisance fits enter a corrected estimator.
% The observed data and the true target remain fixed throughout the loop.
% The target's displayed motion has a quadratic nuisance contribution plus
% a small linear finite-sample contribution and a fixed sampling offset.
% It is NOT a universal finite-sample formula or a quantitative comparison
% between the units of the function plot and the target axis.
\pgfmathdeclarefunction{baseline}{1}{%
  \pgfmathparse{2.67+.40*sin(65*#1)+.13*#1}}
\pgfmathdeclarefunction{direction}{1}{%
  \pgfmathparse{.62*(.32+.68*cos(42*#1))+.11*sin(75*#1)}}

\newcommand{\robustframe}[1]{%
\pgfmathsetmacro{\perturb}{sin(7.5*#1)}
\pgfmathsetmacro{\estimatex}{13.62+.28*\perturb*\perturb+.025*\perturb}
\begin{tikzpicture}[
  x=1cm,y=1cm,text=ink,font=\fontsize{14}{17}\selectfont,
  >={Latex[length=3mm,width=2.1mm]}
]
  \path[use as bounding box] (0,0) rectangle (16.5,5.5);
  \fill[white] (0,0) rectangle (16.5,5.5);

  \node at (3.1,4.78) {Nuisance estimates $\widehat\eta$};
  \node[text=muted,font=\fontsize{10.5}{13}\selectfont]
    at (3.1,4.3) {Outcome or treatment models};
  \node at (13.45,4.78) {Corrected estimate $\widehat\theta$};

  % Nearby functions are possible nuisance fits, not a confidence band.
  \draw[muted!65,line width=.8pt,->] (.85,1.25) -- (5.4,1.25);
  \draw[muted!65,line width=.8pt,->] (.85,1.25) -- (.85,4.22);
  \node[text=muted,anchor=north,font=\fontsize{10.5}{13}\selectfont]
    at (3.1,1.15) {Covariates};
  \foreach \shift in {-1,-.5,0,.5,1}{
    \draw[blue!15,line width=1.05pt]
      plot[domain=-2.2:2.2,samples=55]
        ({3.1+.94*\x},{baseline(\x)+\shift*direction(\x)});
  }
  \draw[blue!65,line width=2pt]
    plot[domain=-2.2:2.2,samples=70]
      ({3.1+.94*\x},{baseline(\x)+\perturb*direction(\x)});

  % One map, with all data held fixed while the nuisance fit changes.
  \draw[ink!75,line width=1.7pt,->] (6.05,2.7) -- (10.12,2.7);
  \node[draw=black!55,fill=green!12,rounded corners=3pt,align=center,
    inner xsep=10pt,inner ysep=8pt] at (8.08,3.72)
    {Orthogonal\\bias correction};
  \node[text=muted,font=\fontsize{10.5}{13}\selectfont]
    at (8.08,2.06) {Data held fixed};

  % A stable scalar target estimate. Faint points show its small movement
  % across the illustrated fits; they are not confidence limits.
  \draw[muted!65,line width=.8pt,->] (11.1,2.7) -- (15.8,2.7);
  \node[star,star points=5,star point ratio=2.25,
    fill=red!80,draw=red!75!black,minimum size=10pt,inner sep=.35pt]
    at (13.15,2.7) {};
  \node[anchor=north,font=\fontsize{10.5}{13}\selectfont]
    at (13.15,2.38) {True target};
  \foreach \shift in {-1,-.5,0,.5,1}{
    \fill[blue!15] ({13.62+.28*\shift*\shift+.025*\shift},2.7)
      circle (2.3pt);
  }
  % The constant halo highlights the current estimate, not its uncertainty.
  \fill[blue!15] (\estimatex,2.7) circle (5pt);
  \fill[blue!80] (\estimatex,2.7) circle (2.7pt);
  \node[anchor=south,inner sep=2pt] at (\estimatex,2.95)
    {$\widehat\theta$};
\end{tikzpicture}%
}
