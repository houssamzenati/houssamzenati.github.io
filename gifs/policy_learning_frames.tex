\documentclass[tikz,border=0pt,multi]{standalone}
\usepackage{tikz}
\usetikzlibrary{
    arrows.meta,
    positioning,
    calc,
    shapes.geometric
}
\renewcommand{\familydefault}{\sfdefault}

% A quadratic risk supplies BOTH its contours and an exact gradient flow:
% R(x,y) = ((x-targetx)/riskxscale)^2/2
%        + ((y-targety)/riskyscale)^2/2.
% This is an illustrative geometry, not a model of a particular algorithm.
\newcommand{\targetx}{5.75}
\newcommand{\targety}{3.35}
\newcommand{\startx}{1.45}
\newcommand{\starty}{1.45}
\newcommand{\riskxscale}{1.35}
\newcommand{\riskyscale}{0.78}
\newcommand{\flowhorizon}{4.12}
\pgfmathdeclarefunction{policyx}{1}{%
    \pgfmathparse{\targetx+(\startx-\targetx)*
        exp(-\flowhorizon*#1/(\riskxscale*\riskxscale))}%
}
\pgfmathdeclarefunction{policyy}{1}{%
    \pgfmathparse{\targety+(\starty-\targety)*
        exp(-\flowhorizon*#1/(\riskyscale*\riskyscale))}%
}

% ------------------------------------------------------------
% Draw one animated frame
% ------------------------------------------------------------
\newcommand{\policyframe}[1]{%
\begin{tikzpicture}[x=1cm,y=1cm,>=Latex,font=\small]
    % Fixed canvas prevents motion-induced cropping and keeps the original 2:1 format.
    \path[use as bounding box] (-0.4,-1.45) rectangle (15.6,6.55);
    \fill[white] (-0.4,-1.45) rectangle (15.6,6.55);

    % --------------------------------------------------------
    % Policy-space geometry
    % --------------------------------------------------------
    \coordinate (centerPi) at (4.40,2.55);
    \coordinate (pstar)    at (\targetx,\targety);

    % Large policy class
    \draw[
        fill=gray!10,
        draw=black!30,
        line width=0.8pt
    ]
    (centerPi) ellipse [
        x radius=4.20,
        y radius=2.25
    ];

    % Keep the class label above the entire policy-improvement path.
    \node[
        anchor=west,
        font=\bfseries\large
    ] at (2.00,4.10)
    {Policy class $\Pi$};

    % --------------------------------------------------------
    % Objective contours, entirely contained in Pi
    % --------------------------------------------------------
    \foreach \level in {1,0.68,0.36}{
        \pgfmathsetmacro{\rx}{\riskxscale*\level}
        \pgfmathsetmacro{\ry}{\riskyscale*\level}
        \draw[
            orange!55,
            line width=0.85pt,
            opacity=0.85
        ]
        (pstar) ellipse [
            x radius=\rx,
            y radius=\ry
        ];
    }

    % --------------------------------------------------------
    % The continuous trail is a gradient-flow orbit; its marked samples also
    % suggest successive policy updates. These are not deployment timestamps.
    % Smoothstep changes animation pacing only, not the geometric orbit.
    % The endpoint remains visibly distinct from the best-in-class comparator.
    % --------------------------------------------------------
    \pgfmathsetmacro{\elapsed}{(#1-1)/60}
    \pgfmathsetmacro{\progress}{3*\elapsed^2-2*\elapsed^3}
    \coordinate (initial) at ({policyx(0)},{policyy(0)});
    \coordinate (current) at ({policyx(\progress)},{policyy(\progress)});

    \ifnum#1>1
        \draw[blue!65,line width=1.15pt,line cap=round,line join=round]
            plot[domain=0:\progress,variable=\s,samples=65]
            ({policyx(\s)},{policyy(\s)});
    \fi

    % Sparse markers keep the geometric illustration easy to read.
    \foreach \mark in {0,0.25,0.50,0.75}{
        \pgfmathtruncatemacro{\reached}{\progress>\mark}
        \ifnum\reached=1
            \fill[blue!65,opacity=0.75]
                ({policyx(\mark)},{policyy(\mark)}) circle (2pt);
        \fi
    }
    \foreach \mark in {0.14,0.38,0.64}{
        \pgfmathtruncatemacro{\reached}{\progress>\mark+0.025}
        \ifnum\reached=1
            \draw[blue!65,line width=1.15pt,-{Latex[length=1.5mm]}]
                plot[domain=\mark-0.025:\mark+0.025,variable=\s,samples=7]
                ({policyx(\s)},{policyy(\s)});
        \fi
    }

    % A fixed halo highlights the current rule; it is not a confidence set.
    \fill[blue!15] (current) circle (5pt);
    \fill[blue!80] (current) circle (2.7pt);
    \ifnum#1>1
        \node[above right=3pt,fill=gray!10,inner sep=1pt]
            at (current) {$\pi$};
    \fi
    \node[below right=2pt] at (initial) {$\pi_0$};

    % --------------------------------------------------------
    % Best-in-class intervention rule, distinct from the learned iterate
    % --------------------------------------------------------
    \node[
        star,
        star points=5,
        star point ratio=2.25,
        fill=red!80,
        draw=red!75!black,
        minimum size=10pt,
        inner sep=0.35pt
    ] (opt) at (pstar) {};

    \node[
        above right=1pt
    ] at (opt)
    {$\pi^\star$};

    % --------------------------------------------------------
    % A complete learning path from existing data, with deployment as an
    % optional extension. Dashed edges collect additional data sequentially.
    % --------------------------------------------------------
    \begin{scope}[shift={(12.35,0.78)}]
        \node[
            draw,rounded corners=3pt,fill=green!12,
            minimum width=4.25cm,minimum height=0.95cm
        ] (data) at (0,3.15)
        {Logged actions and outcomes};

        \node[
            draw,rounded corners=3pt,fill=green!12,
            minimum width=4.25cm,minimum height=0.95cm
        ] (learn) at (0,1.70)
        {Learn an intervention rule $\pi$};

        \node[
            draw=black!55,dashed,rounded corners=3pt,fill=green!5,
            minimum width=4.25cm,minimum height=0.95cm,align=center
        ] (deploy) at (0,0.25)
        {Deploy $\pi$\\[-1pt]{\footnotesize sequential setting}};

        \draw[-{Latex[length=2.2mm]}] (data) -- (learn);
        \draw[dashed,black!60,-{Latex[length=2.2mm]}]
            (learn) -- (deploy);
        \draw[dashed,black!60,-{Latex[length=2.2mm]},rounded corners=5pt]
            (deploy.west) -- (-2.90,0.25) -- (-2.90,3.15) -- (data.west);
        \node[align=center,font=\footnotesize,text=black!70,
            fill=white,inner sep=3pt] at (-2.90,1.70)
            {New\\feedback};
    \end{scope}

    \node[text=black!75] at (7.35,-0.45)
        {Policy improvement through a continuous flow or successive updates.};

\end{tikzpicture}
}
