\documentclass[tikz,border=8pt,multi]{standalone}
\usepackage{tikz}
\usetikzlibrary{
    arrows.meta,
    positioning,
    calc,
    shapes.geometric
}
\renewcommand{\familydefault}{\sfdefault}

% ------------------------------------------------------------
% Draw one animated frame
% ------------------------------------------------------------
\newcommand{\policyframe}[1]{%
\begin{tikzpicture}[x=1cm,y=1cm,>=Latex,font=\small]

    % --------------------------------------------------------
    % Policy-space geometry
    % --------------------------------------------------------
    \coordinate (centerPi) at (4.40,2.55);
    \coordinate (pstar)    at (5.75,3.35);

    % Large policy class
    \draw[
        fill=gray!10,
        draw=black!30,
        line width=0.8pt
    ]
    (centerPi) ellipse [
        x radius=4.20,
        y radius=2.25
    ];

    % Label further left, but clearly inside Pi
    \node[
        anchor=west,
        font=\bfseries\large
    ] at (0.85,3.55)
    {Policy class $\Pi$};

    % --------------------------------------------------------
    % Objective contours, entirely contained in Pi
    % --------------------------------------------------------
    \foreach \rx/\ry in {
        1.35/0.78,
        0.92/0.53,
        0.48/0.28
    }{
        \draw[
            orange!55,
            line width=0.85pt,
            opacity=0.85
        ]
        (pstar) ellipse [
            x radius=\rx,
            y radius=\ry
        ];
    }

    % --------------------------------------------------------
    % Policy iterates
    %
    % The trajectory initially moves rapidly, then bends along
    % the objective geometry and takes progressively smaller
    % steps near pi*.
    % --------------------------------------------------------
    \coordinate (p1)  at (1.55,1.65);
    \coordinate (p2)  at (2.10,1.45);
    \coordinate (p3)  at (2.65,1.68);
    \coordinate (p4)  at (3.20,2.05);
    \coordinate (p5)  at (3.70,2.50);
    \coordinate (p6)  at (4.10,2.90);
    \coordinate (p7)  at (4.45,3.20);
    \coordinate (p8)  at (4.85,3.43);
    \coordinate (p9)  at (5.20,3.52);
    \coordinate (p10) at (5.45,3.50);
    \coordinate (p11) at (5.60,3.44);
    \coordinate (p12) at (5.69,3.39);
    \coordinate (p13) at (pstar);

    % --------------------------------------------------------
    % Smooth curved update steps
    % --------------------------------------------------------
    \ifnum#1>1
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p1) to[out=8,in=195] (p2);
    \fi

    \ifnum#1>2
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p2) to[out=15,in=200] (p3);
    \fi

    \ifnum#1>3
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p3) to[out=25,in=210] (p4);
    \fi

    \ifnum#1>4
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p4) to[out=32,in=215] (p5);
    \fi

    \ifnum#1>5
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p5) to[out=38,in=220] (p6);
    \fi

    \ifnum#1>6
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p6) to[out=35,in=215] (p7);
    \fi

    \ifnum#1>7
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p7) to[out=25,in=200] (p8);
    \fi

    \ifnum#1>8
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p8) to[out=12,in=188] (p9);
    \fi

    \ifnum#1>9
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p9) to[out=-3,in=172] (p10);
    \fi

    \ifnum#1>10
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p10) to[out=-8,in=165] (p11);
    \fi

    \ifnum#1>11
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p11) to[out=-12,in=158] (p12);
    \fi

    \ifnum#1>12
        \draw[
            blue!65,
            line width=1.15pt,
            -{Latex[length=1.8mm]}
        ]
        (p12) to[out=-18,in=150] (p13);
    \fi

    % --------------------------------------------------------
    % Previous iterates
    % --------------------------------------------------------
    \foreach \i in {1,...,#1}{
        \ifnum\i<#1
            \fill[
                blue!70,
                opacity=0.42
            ]
            (p\i) circle (1.5pt);
        \fi
    }

    % --------------------------------------------------------
    % Current estimate and confidence ellipsoid
    % --------------------------------------------------------
    \pgfmathsetmacro{\cx}{0.50 - 0.024*(#1-1)}
    \pgfmathsetmacro{\cy}{0.28 - 0.012*(#1-1)}

    \draw[
        dashed,
        blue!70,
        line width=0.9pt,
        rotate around={18:(p#1)}
    ]
    (p#1) ellipse [
        x radius=\cx,
        y radius=\cy
    ];

    \fill[blue!15] (p#1) circle (7pt);
    \fill[blue!80] (p#1) circle (2.7pt);

    \node[
        above left=1pt
    ] at (p#1)
    {$\hat{\pi}_t$};

    \node[
        blue!65!black,
        below right=1pt
    ] at (p#1)
    {$\mathcal C_t$};

    % Initial policy label, now well inside the policy class
    \node[
        below left=0.5pt
    ] at (p1)
    {$\hat{\pi}_0$};

    % --------------------------------------------------------
    % Optimal policy, drawn last so it remains visible when
    % the final iterate reaches it
    % --------------------------------------------------------
    \node[
        star,
        star points=5,
        star point ratio=2.25,
        fill=red!80,
        draw=red!75!black,
        minimum size=10pt,
        inner sep=0.35pt
    ] (opt) at (pstar) {};

    \node[
        above right=1pt
    ] at (opt)
    {$\pi^\star$};

    % --------------------------------------------------------
    % Deployment and update loop
    % --------------------------------------------------------
    \begin{scope}[shift={(12.35,0.78)}]

        \node[
            draw,
            rounded corners=3pt,
            fill=green!12,
            minimum width=4.25cm,
            minimum height=0.95cm
        ] (deploy) at (0,3.15)
        {Deploy $\hat{\pi}_t$};

        \node[
            draw,
            rounded corners=3pt,
            fill=green!12,
            minimum width=4.25cm,
            minimum height=0.95cm
        ] (observe) at (0,1.70)
        {Observe data $D_t$};

        \node[
            draw,
            rounded corners=3pt,
            fill=green!12,
            minimum width=4.25cm,
            minimum height=0.95cm
        ] (update) at (0,0.25)
        {Update $\hat{\pi}_{t+1}$};

        \draw[-{Latex[length=2.2mm]}]
            (deploy) -- (observe);

        \draw[-{Latex[length=2.2mm]}]
            (observe) -- (update);

    \end{scope}

\end{tikzpicture}
}

% One PDF page per animation frame
% ------------------------------------------------------------
\begin{document}

}
