\newcommand{\basepanel}[3]{%

    \node[font=\small,rotate=90,anchor=south] at (-0.06,0.5) {$x_2$};

    \foreach \rx/\ry in {0.33/0.20,0.25/0.15,0.17/0.10,0.09/0.055}{

  \end{scope}

% Keep the original iterate coordinates and frame sequence. Only visual
% styling and sparse historical markers differ from the earlier illustration.
\newcount\pathstep
\newcommand{\drawseg}[4]{%
  \draw[traj] (#1,#2) -- (#3,#4);
  \coordinate (current) at (#3,#4);
  \advance\pathstep by 1
  \pgfmathtruncatemacro{\markstep}{mod(\the\pathstep,4)}
  \ifnum\markstep=0
    \fill[trajblue,opacity=0.75] (#3,#4) circle (2pt);
  \fi
}
\newcommand{\currentpoint}{%
  \fill[blue!15] (current) circle (5pt);
  \fill[blue!80] (current) circle (2.7pt);
}
\newcommand{\rootmarker}{%
  \begin{scope}[x=5.5cm,y=5.5cm]
    \node[star,star points=5,star point ratio=2.25,
      fill=rootred,draw=red!75!black,
      minimum size=10pt,inner sep=0.35pt] (root) at (0.4028,0.5972) {};
    \node[anchor=west,font=\small,fill=panelbg,inner sep=1pt]
      at (0.54,0.72) {$x^\star$};
  \end{scope}
}
