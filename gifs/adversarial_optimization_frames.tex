\newcommand{\basepanel}[3]{%
  \begin{scope}[shift={(#1,#2)}, x=#3, y=#3]
    \fill[panelbg] (0,0) rectangle (1,1);
    \draw[black!45,line width=0.45pt] (0,0) rectangle (1,1);

    \draw[axisline,-{Latex[length=1.45mm]}] (0,0) -- (1.05,0);
    \draw[axisline,-{Latex[length=1.45mm]}] (0,0) -- (0,1.05);

    \foreach \v in {0,0.5,1}{
      \draw[black!40,line width=0.35pt] (\v,0) -- (\v,-0.015);
      \draw[black!40,line width=0.35pt] (0,\v) -- (-0.015,\v);
    }

    \node[font=\scriptsize,anchor=north] at (0,-0.035) {$0$};
    \node[font=\scriptsize,anchor=north] at (0.5,-0.035) {$0.5$};
    \node[font=\scriptsize,anchor=north] at (1,-0.035) {$1$};

    \node[font=\small,anchor=north] at (0.5,-0.115) {$x_1$};
    \node[font=\small,rotate=90,anchor=south] at (-0.125,0.5) {$x_2$};

    % Global saddle contours.
    \draw[contour] (0.04,0.91) .. controls (0.23,0.79) and (0.38,0.62) .. (0.52,0.50)
                   .. controls (0.65,0.38) and (0.78,0.24) .. (0.96,0.10);

      \draw[contour,rotate around={23:(0.403,0.597)}]
        (0.403,0.597) ellipse [x radius=\rx, y radius=\ry];
    }

    \draw[black!20] (0.18,0.82) -- (0.88,0.12);
    \draw[black!20] (0.14,0.18) -- (0.86,0.90);

    \node[
      star,
      star points=5,
      star point ratio=2.15,
      fill=rootred,
      draw=black,
      line width=0.25pt,
      minimum size=5.8pt,
      inner sep=0pt
    ] at (0.4028,0.5972) {};
    \node[anchor=south west,text=rootred,font=\scriptsize] at (0.42,0.605) {$x^\star$};

    \draw[trajblue,line width=0.8pt] (0.2800,0.5500) circle (3.0pt/#3);
    \node[anchor=west,text=trajblue,font=\scriptsize] at (0.295,0.550) {$x_0$};
  \end{scope}
}
